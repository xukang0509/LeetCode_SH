    class Solution {
        public int missingNumber(int[] nums) {
            int n = nums.length;
            int[] hash = new int[n + 1];
            for (int num : nums) {
                hash[num]++;
            }
            for (int i = 0; i < hash.length; i++) {
                if (hash[i] == 0) return i;
            }
            throw new RuntimeException();
        }
    }
