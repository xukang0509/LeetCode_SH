
    /**
     * Definition for a binary tree node.
     * public class TreeNode {
     *     int val;
     *     TreeNode left;
     *     TreeNode right;
     *     TreeNode() {}
     *     TreeNode(int val) { this.val = val; }
     *     TreeNode(int val, TreeNode left, TreeNode right) {
     *         this.val = val;
     *         this.left = left;
     *         this.right = right;
     *     }
     * }
     */
    class Solution {
        public boolean hasPathSum(TreeNode root, int targetSum) {
            if (root == null) return false;
            return hasPathSum(root, targetSum, root.val);
        }

        private boolean hasPathSum(TreeNode root, int targetSum, int pathSum) {
            if (root.left == null && root.right == null) {
                return targetSum == pathSum;
            }
            if (root.left != null) {
                boolean l = hasPathSum(root.left, targetSum, pathSum + root.left.val);
                if (l) return l;
            }
            if (root.right != null) {
                boolean r = hasPathSum(root.right, targetSum, pathSum + root.right.val);
                if (r) return r;
            }
            return false;
        }
    }
