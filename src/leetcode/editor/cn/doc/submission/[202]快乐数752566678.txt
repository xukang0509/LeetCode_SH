    class Solution {
        public boolean isHappy(int n) {
            Set<Integer> set = new HashSet<>();
            while (n != 1 && !set.contains(n)) {
                set.add(n);
                n = getNextNum(n);
            }
            return n == 1;
        }

        private int getNextNum(int n) {
            int res = 0;
            while (n > 0) {
                int i = n % 10;
                res += i * i;
                n /= 10;
            }
            return res;
        }
    }
