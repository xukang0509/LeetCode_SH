
    /**
     * Definition for a binary tree node.
     * public class TreeNode {
     *     int val;
     *     TreeNode left;
     *     TreeNode right;
     *     TreeNode() {}
     *     TreeNode(int val) { this.val = val; }
     *     TreeNode(int val, TreeNode left, TreeNode right) {
     *         this.val = val;
     *         this.left = left;
     *         this.right = right;
     *     }
     * }
     */
    class Solution {
        public List<String> binaryTreePaths(TreeNode root) {
            List<String> result = new ArrayList<>();
            if (root == null) return result;
            List<String> builder = new ArrayList<>();
            builder.add(String.valueOf(root.val));
            addString(root, builder, result);
            return result;
        }

        private void addString(TreeNode node, List<String> builder, List<String> result) {
            if (node.left == null && node.right == null) {
                result.add(String.join("->", builder));
            }
            if (node.left != null) {
                builder.add(String.valueOf(node.left.val));
                addString(node.left, builder, result);
                builder.remove(builder.size() - 1);
            }
            if (node.right != null) {
                builder.add(String.valueOf(node.right.val));
                addString(node.right, builder, result);
                builder.remove(builder.size() - 1);
            }
        }
    }
