class Solution {
    public int evalRPN(String[] tokens) {
        Stack<Integer> stack = new Stack<>();
        int sum;
        for (String str : tokens) {
            if ("+".equals(str) || "-".equals(str) || "*".equals(str) || "/".equals(str)){
                int num2 = stack.pop();
                int num1 = stack.pop();
                switch (str){
                    case "+":
                        num1 += num2;
                        break;
                    case "-":
                        num1 -= num2;
                        break;
                    case "*":
                        num1 *= num2;
                        break;
                    case "/":
                        num1 /= num2;
                        break;
                    default:
                }
                stack.push(num1);
            } else {
                stack.push(Integer.valueOf(str));
            }
        }
        return stack.pop();
    }
}
