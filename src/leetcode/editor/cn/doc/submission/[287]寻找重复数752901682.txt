    class Solution {
        public int findDuplicate(int[] nums) {
            Set<Integer> set = new HashSet<>();
            int res = -1;
            for (int num : nums) {
                if (set.contains(num)) {
                    res = num;
                    break;
                }
                set.add(num);
            }
            return res;
        }
    }
