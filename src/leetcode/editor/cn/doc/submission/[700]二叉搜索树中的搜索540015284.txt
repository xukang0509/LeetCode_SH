
    /**
     * Definition for a binary tree node.
     * public class TreeNode {
     * int val;
     * TreeNode left;
     * TreeNode right;
     * TreeNode() {}
     * TreeNode(int val) { this.val = val; }
     * TreeNode(int val, TreeNode left, TreeNode right) {
     * this.val = val;
     * this.left = left;
     * this.right = right;
     * }
     * }
     */
    class Solution {
        public TreeNode searchBST(TreeNode root, int val) {
            TreeNode node = root;
            while (node != null) {
                if (val > node.val) {
                    node = node.right;
                } else if (val < node.val) {
                    node = node.left;
                } else {
                    break;
                }
            }
            return node;
        }
    }
