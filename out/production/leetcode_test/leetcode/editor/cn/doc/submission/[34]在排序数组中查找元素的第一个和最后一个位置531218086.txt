class Solution {
    public int[] searchRange(int[] nums, int target) {
        int[] res = new int[2];
        res[0] = leftMost(nums, target);
        res[1] = rightMost(nums, target);
        return res;
    }

    public int leftMost(int[] nums, int target) {
        int left = 0, right = nums.length - 1;
        int candidate = -1;
        while (left <= right) {
            int mid = (left + right) >>> 1;
            if (target < nums[mid]) {
                right = mid - 1;
            } else if (nums[mid] < target) {
                left = mid + 1;
            } else {
                candidate = mid;
                right = mid - 1;
            }
        }
        return candidate;
    }

    public int rightMost(int[] nums, int target) {
        int left = 0, right = nums.length - 1;
        int candidate = -1;
        while (left <= right) {
            int mid = (left + right) >>> 1;
            if (target < nums[mid]) {
                right = mid - 1;
            } else if (nums[mid] < target) {
                left = mid + 1;
            } else {
                candidate = mid;
                left = mid + 1;
            }
        }
        return candidate;
    }
}
